  <script>
    let inventoryData = JSON.parse(localStorage.getItem("inventoryData")) || [];
    let isEditing = false;
    let editIndex = -1;

    // Show notification
    function showNotification(message, type = 'success') {
      const notification = document.getElementById('notification');
      notification.innerHTML = `<i class="fas ${getNotificationIcon(type)}"></i> ${message}`;
      notification.className = `notification ${type} show`;
      
      setTimeout(() => {
        notification.classList.remove('show');
      }, 3000);
    }

    function getNotificationIcon(type) {
      switch(type) {
        case 'success': return 'fa-check-circle';
        case 'error': return 'fa-exclamation-circle';
        case 'warning': return 'fa-exclamation-triangle';
        default: return 'fa-info-circle';
      }
    }

    // Update inventory summary
    function updateInventorySummary() {
      const totalItems = document.getElementById('total-items');
      const totalValue = document.getElementById('total-value');
      const lowStock = document.getElementById('low-stock');
      
      totalItems.textContent = inventoryData.length;
      
      const value = inventoryData.reduce((sum, item) => {
        return sum + (parseFloat(item.price) * parseInt(item.quantity));
      }, 0);
      
      totalValue.textContent = `₱${value.toFixed(2)}`;
      
      const lowStockCount = inventoryData.filter(item => parseInt(item.quantity) < 10).length;
      lowStock.textContent = lowStockCount;
    }

    // Initialize page
    document.addEventListener('DOMContentLoaded', function() {
      updateInventorySummary();
      viewInventory(); // Show inventory by default
      
      // Add search functionality
      document.getElementById('search-inventory').addEventListener('input', function(e) {
        const searchTerm = e.target.value.toLowerCase();
        filterInventory(searchTerm);
      });
    });

    function filterInventory(searchTerm) {
      const rows = document.querySelectorAll('#inventory-list tr');
      
      rows.forEach(row => {
        const medicineName = row.querySelector('td:first-child').textContent.toLowerCase();
        if (medicineName.includes(searchTerm)) {
          row.style.display = '';
        } else {
          row.style.display = 'none';
        }
      });
    }

    function openModal() {
      document.getElementById("myModal").style.display = "block";
      document.getElementById("inventoryForm").reset();
      document.getElementById("modal-title").textContent = "Add Inventory Item";
      isEditing = false;
      editIndex = -1;
    }

    function closeModal() {
      document.getElementById("myModal").style.display = "none";
    }

    // Handle form submission for adding/updating items
    document.getElementById("inventoryForm").addEventListener("submit", function(e) {
      e.preventDefault();

      const medicineName = document.getElementById("medicineName").value;
      const quantity = document.getElementById("quantity").value;
      const price = document.getElementById("price").value;

      if (!medicineName || !quantity || !price) {
        showNotification('Please fill all fields!', 'error');
        return;
      }

      const item = {
        medicineName,
        quantity,
        price
      };

      if (isEditing) {
        // Update existing item
        inventoryData[editIndex] = item;
        showNotification(`Item Updated: ${medicineName}`, 'success');
      } else {
        // Add new item
        inventoryData.push(item);
        showNotification(`Item Added: ${medicineName}`, 'success');
      }

      // Save inventory to localStorage
      localStorage.setItem("inventoryData", JSON.stringify(inventoryData));

      closeModal();
      updateInventorySummary();
      viewInventory();
    });

    // View Inventory function to display items
    function viewInventory() {
      const inventoryContent = document.getElementById("inventory-content");
      const inventoryTable = document.getElementById("inventory-table");
      const reportSection = document.getElementById("report-section");

      // Show the inventory table, hide report
      inventoryTable.style.display = "block";
      reportSection.style.display = "none";

      if (inventoryData.length === 0) {
        // Show empty state
        inventoryContent.innerHTML = `
          <div class="empty-state">
            <i class="fas fa-box-open"></i>
            <h3>No Items in Inventory</h3>
            <p>Your inventory is currently empty. Click the "Add New Item" button to start adding medicines to your inventory.</p>
            <button class="btn btn-primary" onclick="openModal()">
              <i class="fas fa-plus"></i> Add First Item
            </button>
          </div>
        `;
        return;
      }

      // Create table for inventory
      let tableHTML = `
        <table class="inventory-table">
          <thead>
            <tr>
              <th>Medicine Name</th>
              <th>Quantity</th>
              <th>Price</th>
              <th>Total Value</th>
              <th>Actions</th>
            </tr>
          </thead>
          <tbody id="inventory-list">
      `;

      // Add inventory items to the table
      inventoryData.forEach((item, index) => {
        const totalValue = (parseFloat(item.price) * parseInt(item.quantity)).toFixed(2);
        tableHTML += `
          <tr class="fade-in-right" style="animation-delay: ${index * 0.05}s">
            <td>${item.medicineName}</td>
            <td>${item.quantity}</td>
            <td>₱${parseFloat(item.price).toFixed(2)}</td>
            <td>₱${totalValue}</td>
            <td>
              <div class="action-btns">
                <button class="table-btn edit-btn" onclick="editItem(${index})">
                  <i class="fas fa-edit"></i> Edit
                </button>
                <button class="table-btn delete-btn" onclick="deleteItem(${index})">
                  <i class="fas fa-trash-alt"></i> Delete
                </button>
              </div>
            </td>
          </tr>
        `;
      });

      tableHTML += `
          </tbody>
        </table>
      `;

      inventoryContent.innerHTML = tableHTML;
    }
